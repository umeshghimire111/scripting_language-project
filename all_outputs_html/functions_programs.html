5. Greatest Among Three Numbers

No Arguments | No Return 
function findGreatest() {
    var num1 = parseInt(prompt("Enter first number:"));
    var num2 = parseInt(prompt("Enter second number:"));
    var num3 = parseInt(prompt("Enter third number:"));
    var greatest = num1;
    if (num2 > greatest) greatest = num2;
    if (num3 > greatest) greatest = num3;
    return "The greatest number is " + greatest;
}

6. Smallest Among Three Numbers

No Arguments | Returns Value 
function findSmallest() {
    var num1 = parseInt(prompt("Enter first number:"));
    var num2 = parseInt(prompt("Enter second number:"));
    var num3 = parseInt(prompt("Enter third number:"));
    return Math.min(num1, num2, num3);
}

7. Calculator (Switch Case)

With Arguments | No Return 
function calculator(num1, num2, operation) {
    switch (operation) {
        case '+': return "Result: " + (num1 + num2);
        case '-': return "Result: " + (num1 - num2);
        case '*': return "Result: " + (num1 * num2);
        case '/': return "Result: " + (num1 / num2);
        default: return "Invalid operation.";
    }
}

9. Check Palindrome Number

No Arguments | No Return 
function checkPalindrome() {
    var num = prompt("Enter a number:");
    var reverseNum = num.split('').reverse().join('');
    if (num === reverseNum) {
        return num + " is a palindrome.";
    } else {
        return num + " is not a palindrome.";
    }
}

10. Check Armstrong Number

No Arguments | Returns Value 
function isArmstrong() {
    var num = prompt("Enter a number:");
    var sum = 0;
    var temp = num;
    while (temp > 0) {
        var digit = temp % 10;
        sum += Math.pow(digit, 3);
        temp = Math.floor(temp / 10);
    }
    return sum == num;
}

11. Factorial of a Number

With Arguments | No Return 
function factorial(num) {
    var fact = 1;
    for (var i = 1; i <= num; i++) {
        fact *= i;
    }
    return "The factorial of " + num + " is " + fact;
}

12. Factors of a Number

With Arguments | Returns Value 
function getFactors(num) {
    var factors = [];
    for (var i = 1; i <= num; i++) {
        if (num % i === 0) factors.push(i);
    }
    return factors;
}

13. Fibonacci Series

With Arguments | No Return 
function fibonacci(n) {
    var a = 0, b = 1, nextTerm;
    var res = "Fibonacci Series: " + a + ", " + b;
    for (var i = 1; i <= n - 2; i++) {
        nextTerm = a + b;
        res += ", " + nextTerm;
        a = b; b = nextTerm;
    }
    return res;
}
